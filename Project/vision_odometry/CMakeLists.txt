cmake_minimum_required(VERSION 3.16)
project(validated_visual_odometry LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs features2d calib3d videoio)
add_library(vo_core src/odometry.cpp)
target_include_directories(vo_core PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(vo_core PUBLIC ${OpenCV_LIBS})
target_compile_options(vo_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(vision_odometry src/main.cpp)
target_link_libraries(vision_odometry PRIVATE vo_core)
include(CTest)
if(BUILD_TESTING)
 add_executable(test_geometry tests/test_geometry.cpp)
 target_link_libraries(test_geometry PRIVATE vo_core)
 add_test(NAME geometry COMMAND test_geometry ${CMAKE_CURRENT_SOURCE_DIR}/config/synthetic.json)
endif()
